# MoneyApp Fase 1 (Base) — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App Expo que corre en Android e iOS con registro/login, primer uso (moneda principal + primera cuenta + grupos y categorías por defecto) y Ajustes para administrar cuentas, grupos y categorías.

**Architecture:** App Expo (Expo Router, TypeScript) en la raíz del repo; rutas en `src/app`, lógica pura en `src/domain` (con pruebas Jest), acceso a datos en `src/data`, componentes visuales en `src/ui`. Backend Supabase: migraciones SQL en `supabase/migrations` con RLS en todas las tablas y una función `complete_onboarding` atómica; pruebas pgTAP en `supabase/tests`.

**Tech Stack:** Expo SDK 57, Expo Router, TypeScript, Jest (`jest-expo`), Supabase (Auth, Postgres, RLS, CLI vía `npx supabase`), `@supabase/supabase-js`, AsyncStorage, `expo-blur`, `expo-haptics`, `expo-symbols`, `@expo-google-fonts/inter`.

**Spec:** `docs/superpowers/specs/2026-10-14-moneyapp-design.md` (secciones 2, 3, 4.5, 5 "Primer uso", 7, 8, 9, 10 fase 1).

## Global Constraints

- Toda la interfaz en **español**.
- Montos como **enteros en la unidad mínima**. Monedas soportadas: `COP` (0 decimales — decisión: el peso no usa centavos en la práctica), `USD`, `EUR`, `MXN` (2 decimales).
- **RLS activado en todas las tablas**; cada usuario solo ve y modifica sus filas.
- Estilo Apple **idéntico en iOS y Android**: nada de Material Design. Fondos `#F2F2F7`/`#000000`, tarjetas radio 16, acento `#007AFF`/`#0A84FF`.
- Fuente del sistema en iOS; **Inter** en Android.
- Emojis como íconos de categorías.
- Colores de categorías: solo la paleta de sistema de Apple definida en `src/domain/catalog.ts`.
- Nunca subir claves: `.env` está en `.gitignore`; solo se sube `.env.example`.
- Todo el trabajo en la rama `feature/fase-1-base`, creada desde `develop`.

```ts
export type CurrencyCode = 'COP' | 'USD' | 'EUR' | 'MXN';

export const CURRENCIES: Record<CurrencyCode, { name: string; decimals: number }> = {
  COP: { name: 'Peso colombiano', decimals: 0 },
  USD: { name: 'Dólar estadounidense', decimals: 2 },
  EUR: { name: 'Euro', decimals: 2 },
  MXN: { name: 'Peso mexicano', decimals: 2 },
};

const grouped = (sep: string) => new RegExp(`^\\d{1,3}(\\${sep}\\d{3})+$`);

/** Convierte texto escrito por el usuario a unidades mínimas. null si no es válido. */
export function parseAmount(text: string, currency: CurrencyCode): number | null {
  const { decimals } = CURRENCIES[currency];
  let s = text.replace(/[\s$]/g, '');
  const negative = s.startsWith('-');
  if (negative) s = s.slice(1);
  if (!/^\d[\d.,]*$/.test(s)) return null;

  let intPart = s;
  let frac = '';
  const hasDot = s.includes('.');
  const hasComma = s.includes(',');
  if (hasDot && hasComma) {
    // El último separador es el decimal; el otro, de miles.
    const dec = s.lastIndexOf('.') > s.lastIndexOf(',') ? '.' : ',';
    const thousands = dec === '.' ? ',' : '.';
    const i = s.lastIndexOf(dec);
    intPart = s.slice(0, i);
    frac = s.slice(i + 1);
    if (!grouped(thousands).test(intPart)) return null;
    intPart = intPart.split(thousands).join('');
  } else if (hasDot || hasComma) {
    const sep = hasDot ? '.' : ',';
    const parts = s.split(sep);
    if (grouped(sep).test(s)) intPart = parts.join('');
    else if (parts.length === 2) [intPart, frac] = parts;
    else return null;
  }

  if (!/^\d+$/.test(intPart) || !/^\d*$/.test(frac) || frac.length > decimals) return null;
  const value = Number(intPart) * 10 ** decimals + Number(frac.padEnd(decimals, '0') || 0);
  return negative ? -value : value;
}

export function formatMoney(minor: number, currency: CurrencyCode): string {
  const { decimals } = CURRENCIES[currency];
  return new Intl.NumberFormat('es-CO', {
    style: 'currency',
    currency,
    minimumFractionDigits: decimals,
    maximumFractionDigits: decimals,
  }).format(minor / 10 ** decimals);
}

/** Monto sin separadores de miles y con coma decimal, para precargar un campo. */
export function amountToInput(minor: number, currency: CurrencyCode): string {
  const { decimals } = CURRENCIES[currency];
  const sign = minor < 0 ? '-' : '';
  const abs = Math.abs(minor);
  if (decimals === 0) return `${sign}${abs}`;
  const int = Math.floor(abs / 10 ** decimals);
  const frac = String(abs % 10 ** decimals).padStart(decimals, '0');
  return `${sign}${int},${frac}`;
}
```

```ts
export function validateName(name: string): string | null {
  const trimmed = name.trim();
  if (!trimmed) return 'Escribe un nombre';
  if (trimmed.length > 40) return 'Máximo 40 caracteres';
  return null;
}

export function validateCredentials(email: string, password: string): string | null {
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim())) return 'Escribe un correo válido';
  if (password.length < 8) return 'La contraseña debe tener mínimo 8 caracteres';
  return null;
}

export function validateEmoji(emoji: string): string | null {
  const trimmed = emoji.trim();
  if (!trimmed) return 'Elige un emoji';
  if (trimmed.length > 16) return 'Usa un solo emoji';
  return null;
}

export type GroupAllocation = {
  id?: string;
  allocation_type: string; // 'percent' | 'amount'
  allocation_value: number;
};

export function percentTotal(groups: GroupAllocation[]): number {
  return groups
    .filter((g) => g.allocation_type === 'percent')
    .reduce((sum, g) => sum + g.allocation_value, 0);
}

export function validateGroupAllocation(
  group: GroupAllocation,
  allGroups: GroupAllocation[],
): string | null {
  const v = group.allocation_value;
  if (group.allocation_type === 'amount') return v < 0 ? 'El monto no puede ser negativo' : null;
  if (!Number.isInteger(v) || v < 0 || v > 100)
    return 'El porcentaje debe ser un número entero entre 0 y 100';
  const total = percentTotal(allGroups.filter((g) => !group.id || g.id !== group.id)) + v;
  return total > 100 ? `Los porcentajes sumarían ${total} %. El máximo es 100 %` : null;
}
```

```ts
const AUTH_MESSAGES: Record<string, string> = {
  invalid_credentials: 'Correo o contraseña incorrectos',
  user_already_exists: 'Ya existe una cuenta con ese correo',
  email_exists: 'Ya existe una cuenta con ese correo',
  weak_password: 'La contraseña es muy débil (mínimo 8 caracteres)',
  email_not_confirmed: 'Confirma tu correo antes de entrar',
  over_request_rate_limit: 'Demasiados intentos. Espera un momento',
  over_email_send_rate_limit: 'Demasiados intentos. Espera un momento',
};

/** Convierte cualquier error (Supabase, red, etc.) en un mensaje para el usuario. */
export function errorMessage(err: unknown): string {
  const e = (err ?? {}) as { code?: string; message?: string };
  if (e.code && AUTH_MESSAGES[e.code]) return AUTH_MESSAGES[e.code];
  const msg = e.message ?? '';
  if (/network request failed|failed to fetch/i.test(msg))
    return 'Sin conexión. Revisa tu internet e intenta de nuevo';
  if (msg.includes('already_onboarded')) return 'Ya completaste la configuración inicial';
  return 'Algo salió mal. Intenta de nuevo';
}
```

```tsx
import { Platform, Text, TextProps } from 'react-native';
import { useTheme } from '@/theme/colors';

type Weight = 'regular' | 'semibold' | 'bold';
const VARIANTS = {
  largeTitle: { fontSize: 34, lineHeight: 41, weight: 'bold' as Weight },
  title: { fontSize: 22, lineHeight: 28, weight: 'semibold' as Weight },
  headline: { fontSize: 17, lineHeight: 22, weight: 'semibold' as Weight },
  body: { fontSize: 17, lineHeight: 22, weight: 'regular' as Weight },
  footnote: { fontSize: 13, lineHeight: 18, weight: 'regular' as Weight },
};
const INTER: Record<Weight, string> = {
  regular: 'Inter_400Regular',
  semibold: 'Inter_600SemiBold',
  bold: 'Inter_700Bold',
};
const IOS_WEIGHT = { regular: '400', semibold: '600', bold: '700' } as const;

// En iOS la fuente del sistema (SF Pro); en Android, Inter.
export function fontFor(weight: Weight) {
  return Platform.OS === 'ios' ? { fontWeight: IOS_WEIGHT[weight] } : { fontFamily: INTER[weight] };
}

type Props = TextProps & {
  variant?: keyof typeof VARIANTS;
  color?: 'text' | 'secondary' | 'accent' | 'red' | 'green';
};

export function AppText({ variant = 'body', color = 'text', style, ...props }: Props) {
  const c = useTheme();
  const { weight, ...size } = VARIANTS[variant];
  return <Text style={[size, fontFor(weight), { color: c[color] }, style]} {...props} />;
}
```

```tsx
import type { Session } from '@supabase/supabase-js';
import { createContext, ReactNode, useCallback, useContext, useEffect, useState } from 'react';
import type { Tables } from '@/lib/database.types';
import { supabase } from '@/lib/supabase';

export type Profile = Tables<'profiles'>;
type Status = 'loading' | 'error' | 'signedOut' | 'onboarding' | 'ready';
type AuthState = {
  status: Status;
  session: Session | null;
  profile: Profile | null;
  refreshProfile: () => Promise<void>;
};

const AuthContext = createContext<AuthState | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [profile, setProfile] = useState<Profile | null>(null);
  const [sessionLoaded, setSessionLoaded] = useState(false);
  const [failed, setFailed] = useState(false);

  const loadProfile = useCallback(async (s: Session | null) => {
    setFailed(false);
    if (!s) return setProfile(null);
    const { data, error } = await supabase.from('profiles').select('*').eq('id', s.user.id).single();
    if (error) setFailed(true);
    setProfile(data);
  }, []);

  useEffect(() => {
    supabase.auth.getSession().then(async ({ data }) => {
      setSession(data.session);
      await loadProfile(data.session);
      setSessionLoaded(true);
    });
    const { data: sub } = supabase.auth.onAuthStateChange((_event, s) => {
      setSession(s);
      // Supabase recomienda no llamar a la API dentro del callback directamente.
      setTimeout(() => loadProfile(s), 0);
    });
    return () => sub.subscription.unsubscribe();
  }, [loadProfile]);

  const status: Status = !sessionLoaded
    ? 'loading'
    : !session
      ? 'signedOut'
      : failed
        ? 'error'
        : profile?.id !== session.user.id
          ? 'loading'
          : profile.main_currency
            ? 'ready'
            : 'onboarding';

  const refreshProfile = useCallback(() => loadProfile(session), [loadProfile, session]);

  return (
    <AuthContext.Provider value={{ status, session, profile, refreshProfile }}>{children}</AuthContext.Provider>
  );
}

export function useAuth(): AuthState {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth debe usarse dentro de AuthProvider');
  return ctx;
}
```

```ts
import { useFocusEffect } from 'expo-router';
import { useCallback, useState } from 'react';
import { errorMessage } from '@/domain/errors';

/** Carga datos cada vez que la pantalla gana foco (p. ej. al volver de un formulario). */
export function useFocusQuery<T>(fn: () => Promise<T>) {
  const [data, setData] = useState<T | null>(null);
  const [error, setError] = useState<string | null>(null);
  useFocusEffect(
    useCallback(() => {
      let alive = true;
      fn()
        .then((d) => alive && (setData(d), setError(null)))
        .catch((e) => alive && setError(errorMessage(e)));
      return () => {
        alive = false;
      };
    }, [fn]),
  );
  return { data, error };
}
```

```ts
import { supabase } from '@/lib/supabase';

export type Account = Tables<'accounts'>;
export type AccountInput = Pick<TablesInsert<'accounts'>, 'name' | 'type' | 'currency' | 'initial_balance'>;

export async function listAccounts(): Promise<Account[]> {
  const { data, error } = await supabase.from('accounts').select('*').order('created_at');
  if (error) throw error;
  return data;
}

export async function getAccount(id: string): Promise<Account> {
  const { data, error } = await supabase.from('accounts').select('*').eq('id', id).single();
  if (error) throw error;
  return data;
}

export async function saveAccount(input: AccountInput, id?: string): Promise<void> {
  const { error } = id
    ? await supabase.from('accounts').update(input).eq('id', id)
```
